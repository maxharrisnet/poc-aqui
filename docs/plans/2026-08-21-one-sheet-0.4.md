# One sheet (v0.4) Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Fold the watchlist into the inventory: one sheet, one row per title, one tab; delete the watchlist module, endpoint and tab.

**Architecture:** `src/inventory.ts` grows nine appended columns (range `A:Z`) and becomes the only schema. `src/run.ts` gains `sweepRecord`, shared by the `mode=watchlist` stream and by `/api/sourcing`, so a button press and a sweep land on the same number. `api/inventory.ts` absorbs the watchlist's POST (add by release id or text, with the pressing plan and duplicate check). The page loses the Watchlist tab and gains an Add records section, two views and a best-seen line on the Inventory tab.

**Tech Stack:** TypeScript, Node, Google Sheets REST v4 via `src/sheets.ts`, Vercel functions, `node:test` via `tsx`. No new dependencies.

**Spec:** `docs/plans/2026-08-21-one-sheet-0.4-design.md`

/** The pressings a sweep prices for this row. A row added through Watch or Add
 *  carries every pressing of its master; a row typed or seeded by hand has
 *  only the pressing on the shelf, which is the right thing to price until
 *  scripts/enrich-inventory.ts has resolved its master. */
export function sweepReleaseIds(item: Pick<InventoryItem, "releaseId" | "watchedReleaseIds">): number[] {
  if (item.watchedReleaseIds.length > 0) return item.watchedReleaseIds;
  return item.releaseId === null ? [] : [item.releaseId];
}

/** The row that already covers a release: the shelf pressing itself or any
 *  watched pressing of the same album. Inactive rows count, because reviving
 *  one beats appending a duplicate of it. */
export function findByRelease(items: InventoryItem[], releaseId: number): InventoryItem | undefined {
  return items.find((i) => i.releaseId === releaseId || i.watchedReleaseIds.includes(releaseId));
}

```ts
export interface SweepOutcome {
  result: CheckResult;
  /** Pressings actually priced. Fewer than asked when Discogs failed on some. */
  checked: number;
  /** The cheapest copy landed at or under the record's alert price. */
  under: boolean;
  /** The row as written back. */
  item: InventoryItem;
}

/**
 * Prices every watched pressing of one record, keeps the cheapest, and writes
 * what it found back to the row.
 *
 * One function for both callers on purpose. The Check listings button on an
 * inventory row and the scheduled sweep have to land on the same number, or
 * the figure a person saw at the counter is not the figure the alert fired on.
 */
export async function sweepRecord(item: InventoryItem, fx: FxSnapshot): Promise<SweepOutcome> {
  const label = `${item.artist}: ${item.album}`;
  const releaseIds = sweepReleaseIds(item);
  const checkedAt = new Date().toISOString();

  let best: CheckResult | null = null;
  let checkError: string | null = null;
  let checked = 0;

  for (const releaseId of releaseIds) {
    try {
      const result = await costRelease(releaseId, label, fx, item.notes || undefined);
      checked += 1;
      if (result.cost && (!best?.cost || result.cost.totalUsd < best.cost.totalUsd)) {
        best = result;
      }
    } catch (err) {
      // Remember why, but keep trying the other pressings.
      checkError = (err as Error).message;
    }
  }

  if (best) {
    const under =
      best.totalMxn != null && item.maxLandedMxn != null && best.totalMxn <= item.maxLandedMxn;
    // Only on the crossing, not on every sweep of an already-alerted record:
    // a nightly re-alert for the same copy is how a team learns to ignore the
    // channel.
    if (under && item.alertSms && item.status !== "alerted") {
      best.alert = composeAlert(item, best);
    }
    const patch: Partial<InventoryItem> = {
      lastCheckedAt: checkedAt,
      bestLandedMxn: best.totalMxn ?? null,
      bestReleaseId: best.discogsId ?? null,
      status: under ? "alerted" : "watching",
    };
    await safePatch(item.id, patch);
    return { result: best, checked, under, item: { ...item, ...patch } };
  }

  if (checkError) {
    // Could not reach Discogs. Record that we tried; change nothing else.
    // Nulling the stored best price here would destroy a real alert.
    const failed: CheckResult = {
      requested: label,
      matched: false,
      unmatchedReason: checkError,
      available: false,
      numForSale: 0,
    };
    const patch: Partial<InventoryItem> = { lastCheckedAt: checkedAt };
    await safePatch(item.id, patch);
    return { result: failed, checked, under: false, item: { ...item, ...patch } };
  }

  // Genuinely checked, genuinely nothing listed. Or nothing to check at all.
  const none: CheckResult = {
    requested: label,
    matched: false,
    unmatchedReason:
      releaseIds.length === 0
        ? "No Discogs release matched, so there is nothing to price"
        : "No copies listed for any watched pressing",
    available: false,
    numForSale: 0,
  };
  const patch: Partial<InventoryItem> = {
    lastCheckedAt: checkedAt,
    bestLandedMxn: null,
    bestReleaseId: null,
    status: "watching",
  };
  await safePatch(item.id, patch);
  return { result: none, checked, under: false, item: { ...item, ...patch } };
}

/**
 * Sweeps every active record whose watch is on.
 *
 * Writes each record's result back as it completes rather than at the end, so
 * a function timeout costs the tail of the sweep instead of all of it. The
 * sheet doubles as the checkpoint.
 */
export async function runWatched(
  onProgress?: ProgressFn,
  onResult?: (r: CheckResult) => void,
): Promise<RunSummary> {
  const fx = await fetchFxSnapshot();
  const watched = (await listInventoryItems()).filter((i) => i.active && i.watching);
  // Oldest-checked first, never-checked before that, so a capped run always
  // makes progress on the tail instead of re-sweeping the same prefix.
  watched.sort((a, b) => (a.lastCheckedAt ?? "").localeCompare(b.lastCheckedAt ?? ""));
  const items = watched.slice(0, MAX_SWEEP_ITEMS);
  const skipped = watched.length - items.length;
  const results: CheckResult[] = [];

  for (const [i, item] of items.entries()) {
    onProgress?.({ index: i + 1, total: items.length, label: `${item.artist}: ${item.album}` });
    const { result } = await sweepRecord(item, fx);
    results.push(result);
    onResult?.(result);
  }

  if (skipped > 0) {
    const notice: CheckResult = {
      requested: `${skipped} more record${skipped === 1 ? "" : "s"} not checked this run`,
      matched: false,
      unmatchedReason:
        `A sweep is capped at ${MAX_SWEEP_ITEMS} records to stay inside the time limit. ` +
        `The least recently checked are done first, so run it again to continue.`,
      available: false,
      numForSale: 0,
    };
    results.push(notice);
    onResult?.(notice);
  }

  return summarise("watchlist", fx, results);
}
```

import type { IncomingMessage, ServerResponse } from "node:http";
import { sweepRecord } from "../src/run.js";
import { fetchFxSnapshot } from "../src/fx.js";
import { listInventoryItems } from "../src/inventory.js";
import { publicSheetsError } from "../src/sheets.js";

/**
 * Prices one inventory row on demand, for the Sourcing section of that row.
 *
 *   GET /api/sourcing?id=i_abc123
 *
 * Runs the same per-record sweep a scheduled run would, over every watched
 * pressing, and writes the result back to the row. Two Discogs calls per
 * pressing at the client's pacing, so a six-pressing album takes about
 * thirteen seconds. A person is waiting for it, which is why the page says
 * how many pressings it is pricing.
 *
 * Only Discogs returns a real price. eBay, Mercado Libre and Bandcamp are
 * search links rather than results, and are labelled that way in the response
 * so the interface cannot quietly present a search as a listing. When those
 * APIs are built, they move from `links` to `offers` and nothing else changes.
 */

function json(res: ServerResponse, status: number, body: unknown): void {
  res.writeHead(status, { "Content-Type": "application/json", "Cache-Control": "no-store" });
  res.end(JSON.stringify(body));
}

/** Search URLs, not listings. Each one opens the platform with the record's
 *  name already typed in, which is what a person does by hand today. */
function searchLinks(query: string): { platform: string; url: string; note: string }[] {
  const q = encodeURIComponent(query);
  // Mercado Libre puts the search in the path, hyphen-separated, and treats a
  // percent-encoded space as part of the term rather than a separator.
  const slug = query
    .toLowerCase()
    .normalize("NFD")
    .replace(/[\u0300-\u036f]/g, "")
    .replace(/[^a-z0-9]+/g, "-")
    .replace(/^-|-$/g, "");
  return [
    {
      platform: "eBay",
      url: `https://www.ebay.com/sch/i.html?_nkw=${q}+vinyl`,
      note: "Search. The Browse API would return real listings with shipping to Mexico",
    },
    {
      platform: "Mercado Libre",
      url: `https://listado.mercadolibre.com.mx/${slug}-vinilo`,
      note: "Search. Domestic listings, so no import cost at all",
    },
    {
      platform: "Bandcamp",
      url: `https://bandcamp.com/search?q=${q}&item_type=p`,
      note: "Search. No API exists, and no prices are readable programmatically",
    },
  ];
}
